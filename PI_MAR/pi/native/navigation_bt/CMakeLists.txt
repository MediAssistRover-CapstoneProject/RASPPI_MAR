cmake_minimum_required(VERSION 3.16)
project(mar_navigation_bt)
find_package(ament_cmake REQUIRED)
find_package(behaviortree_cpp_v3 REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(nav2_util REQUIRED)
find_package(nav2_regulated_pure_pursuit_controller REQUIRED)
find_package(pluginlib REQUIRED)
find_package(angles REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
add_library(mar_commit_navigation_path_bt_node SHARED commit_navigation_path.cpp)
target_compile_features(mar_commit_navigation_path_bt_node PRIVATE cxx_std_17)
target_compile_definitions(mar_commit_navigation_path_bt_node PRIVATE BT_PLUGIN_EXPORT)
ament_target_dependencies(mar_commit_navigation_path_bt_node behaviortree_cpp_v3 nav_msgs)
add_library(mar_forward_pivot_navigation SHARED forward_pivot_planner.cpp forward_pivot_controller.cpp)
target_compile_features(mar_forward_pivot_navigation PRIVATE cxx_std_17)
ament_target_dependencies(mar_forward_pivot_navigation nav2_core nav2_costmap_2d nav2_util
  nav2_regulated_pure_pursuit_controller pluginlib angles tf2 tf2_geometry_msgs rclcpp_lifecycle)
pluginlib_export_plugin_description_file(nav2_core navigation_plugins.xml)
install(TARGETS mar_commit_navigation_path_bt_node mar_forward_pivot_navigation LIBRARY DESTINATION lib)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_commit_navigation_path test_commit_navigation_path.cpp)
  target_compile_features(test_commit_navigation_path PRIVATE cxx_std_17)
  ament_target_dependencies(test_commit_navigation_path behaviortree_cpp_v3 nav_msgs)
  add_test(NAME commit_path_transaction COMMAND test_commit_navigation_path $<TARGET_FILE:mar_commit_navigation_path_bt_node> ${CMAKE_CURRENT_SOURCE_DIR}/../../config/navigate_committed_path.xml)
endif()
ament_package()
